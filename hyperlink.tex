\documentclass{article}
\usepackage[utf8]{inputenc}
\usepackage{xcolor}
\usepackage[normalem]{ulem}
\usepackage{hyperref}

% Configure hyperref to remove the default colorful boxes
\hypersetup{
    colorlinks=true,
    urlcolor=blue,
}

% Create a custom command for blue underlined links
% Usage: \bluehref{URL}{Text}
\newcommand{\bluehrefuline}[2]{\href{#1}{\textcolor{blue}{\uline{#2}}}}



\begin{document}

Here is a \href{https://www.google.com}{hyperlink to Google} that is blue and has no underline.

Here is a \bluehrefuline{https://www.duckduckgo.com}{hyperlink to DuckDuckGo} that is blue and has underline.
